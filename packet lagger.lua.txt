local function isCarrying()
    local char = LocalPlayer.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return false end
    if hum.WalkSpeed <= 0 then return false end
    for _, child in pairs(char:GetChildren()) do
        local n = child.Name:lower()
        if n:find("brainrot") or n:find("brain") or n:find("animal")
        or n:find("carry") or n:find("stolen") or n:find("held") or n:find("steal") then return true end
    end
    for attrName, attrValue in pairs(char:GetAttributes()) do
        local n = attrName:lower()
        if (n:find("carrying") or n:find("carry") or n:find("stealing")
        or n:find("isstealing") or n:find("hasbrainrot")) and attrValue == true then return true end
    end
    if hum.WalkSpeed > 0 and hum.WalkSpeed <= 25 and hum.WalkSpeed ~= 16 then return true end
    return false
end

local function startE01HeadCountdown()
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("Head") then return end
    local head = char.Head
    if head:FindFirstChild("HeadStealUI") then head.HeadStealUI:Destroy() end
    local bb = Instance.new("BillboardGui")
    bb.Name, bb.Adornee = "HeadStealUI", head
    bb.Size, bb.StudsOffset, bb.AlwaysOnTop = UDim2.new(0,600,0,80), Vector3.new(0,7,0), true
    bb.Parent = head
    local tl = Instance.new("TextLabel", bb)
    tl.Size, tl.BackgroundTransparency = UDim2.new(1,0,1,0), 1
    tl.TextColor3, tl.TextStrokeColor3, tl.TextStrokeTransparency = Color3.fromRGB(255,60,60), Color3.fromRGB(0,0,0), 0
    tl.Font, tl.TextSize, tl.Text = Enum.Font.FredokaOne, 28, ""
    local duration, startTime, finished = 3.0, tick(), false
    local conn
    conn = RunService.RenderStepped:Connect(function()
        if finished then return end
        local remaining = math.max(0, duration - (tick() - startTime))
        if not tl.Parent then finished = true conn:Disconnect() return end
        if remaining > 0 then
            tl.Text = string.format("⚠️ E01 WARNİNG (%.1fs)", remaining)
        else
            finished = true; conn:Disconnect()
            tl.Text, tl.TextColor3 = "⚡ STEAL NOW!", Color3.fromRGB(50,255,130)
            task.spawn(function()
                task.wait(2)
                if bb.Parent then
                    local t = TweenService:Create(tl, TweenInfo.new(0.5), {TextTransparency=1, TextStrokeTransparency=1})
                    t:Play(); t.Completed:Connect(function() pcall(function() bb:Destroy() end) end)
                end
            end)
        end
    end)
end

local function applyAntiLag(state)
    pcall(function()
        if state then
            Lighting.GlobalShadows = false
            Lighting.ShadowSoftness = 0
            Lighting.Brightness = 3
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0
            Lighting.Outlines = false
            pcall(function() Lighting.Technology = Enum.Technology.Compatibility end)
        else
            Lighting.GlobalShadows = St.originalLighting.GlobalShadows or true
            Lighting.ShadowSoftness = St.originalLighting.ShadowSoftness or 0.2
            Lighting.Brightness = St.originalLighting.Brightness or 2
            Lighting.EnvironmentDiffuseScale = St.originalLighting.EnvironmentDiffuseScale or 1
            Lighting.EnvironmentSpecularScale = St.originalLighting.EnvironmentSpecularScale or 1
        end
    end)
end

local function getActiveSoundPair()
    if St.halloweenToggleOn then
        local idx = cfg.halloweenSoundIndex or 0
        if idx == 0 then return nil, nil end
        return K.HALLOWEEN_SOUNDS[idx], K.HALLOWEEN_PITCH[idx]
    else
        local idx = cfg.soundIndex or 0
        if idx == 0 then return nil, nil end
        return K.CLICK_SOUNDS[idx], K.TYPING_PITCH[idx]
    end
end

local function playClick()
    local id, pitch = getActiveSoundPair()
    if not id then return end
    local s = Instance.new("Sound")
    s.SoundId = id
    s.Volume = tonumber(cfg.clickVolume) or 1.2
    if St.halloweenToggleOn and pitch then s.PlaybackSpeed = pitch end
    s.Parent = U.screen; s:Play()
    Debris:AddItem(s, 3)
end

local function playTyping()
    local id, pitch = getActiveSoundPair()
    if not id then return end
    local s = Instance.new("Sound")
    s.SoundId = id
    s.Volume = (tonumber(cfg.clickVolume) or 1.2) * 0.5
    if St.halloweenToggleOn and pitch then s.PlaybackSpeed = pitch * 1.2
    else s.PlaybackSpeed = pitch or 1.6 end
    s.Parent = U.screen; s:Play()
    Debris:AddItem(s, 1)
end

local function attachClickSound(btn)
    if not (btn:IsA("TextButton") or btn:IsA("ImageButton")) then return end
    if btn:GetAttribute("CPSoundAttached") then return end
    btn:SetAttribute("CPSoundAttached", true)
    btn.MouseButton1Click:Connect(function() pcall(playClick) end)
end

    if input.UserInputType ~= Enum.UserInputType.Keyboard then return end
    if not UserInputService:GetFocusedTextBox() then return end
    if K.IGNORE_KEYS[input.KeyCode] then return end
    pcall(playTyping)
end)

    local knob = Instance.new("Frame", sw)
    knob.Size = UDim2.new(0, 18, 0, 18)
    knob.BorderSizePixel = 0
    knob.ZIndex = 9
    Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)
    sw:SetAttribute("ToggleOn", initial)
    if cfgKey then sw:SetAttribute("CfgKey", cfgKey) end
    local function refresh(state, animate)
        local theme = K.THEMES[cfg.themeIndex] or K.THEMES[8]
        local onColor = forceOnColor or theme.main
        local tc = state and onColor or Color3.fromRGB(26, 24, 30)
        local kc = state and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(110, 105, 120)
        local tp = state and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9)
        if animate then
            TweenService:Create(sw, TweenInfo.new(0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), { BackgroundColor3 = tc }):Play()
            TweenService:Create(knob, TweenInfo.new(0.2, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), { Position = tp, BackgroundColor3 = kc }):Play()
        else
            sw.BackgroundColor3 = tc; knob.Position = tp; knob.BackgroundColor3 = kc
        end
    end
    refresh(initial, false)
    sw.MouseButton1Click:Connect(function()
        local ns = not sw:GetAttribute("ToggleOn")
        sw:SetAttribute("ToggleOn", ns)
        refresh(ns, true)
        if onChange then onChange(ns) end
        saveConfig()
    end)
    table.insert(U.toggleSwitches, {
        track = sw, knob = knob,
        refresh = function() refresh(sw:GetAttribute("ToggleOn"), false) end
    })
    return sw
end

function _G.CPL_refreshSoundButtons()
    local theme = K.THEMES[cfg.themeIndex] or K.THEMES[8]
    for i, b in pairs(U.soundButtons) do
        local opt = soundOptions[i]
        local isActive = (opt and opt.value == cfg.soundIndex)
        if St.halloweenToggleOn then
            b.BackgroundColor3 = Color3.fromRGB(18,18,18)
            b.BackgroundTransparency = 0.4
            b.TextColor3 = Color3.fromRGB(90,90,90)
        elseif isActive then
            b.BackgroundColor3 = theme.main
            b.BackgroundTransparency = 0
            b.TextColor3 = Color3.fromRGB(0,0,0)
        else
            b.BackgroundColor3 = Color3.fromRGB(15,15,15)
            b.BackgroundTransparency = 0
            b.TextColor3 = Color3.fromRGB(255,255,255)
        end
    end
    for i, b in pairs(U.hSoundButtons) do
        local opt = hSoundOptions[i]
        local isActive = (opt and opt.value == cfg.halloweenSoundIndex)
        if not St.halloweenToggleOn then
            b.BackgroundColor3 = Color3.fromRGB(18,18,18)
            b.BackgroundTransparency = 0.4
            b.TextColor3 = Color3.fromRGB(90,90,90)
        elseif isActive then
            b.BackgroundColor3 = Color3.fromRGB(255,140,0)
            b.BackgroundTransparency = 0
            b.TextColor3 = Color3.fromRGB(0,0,0)
        else
            b.BackgroundColor3 = Color3.fromRGB(15,15,15)
            b.BackgroundTransparency = 0
            b.TextColor3 = Color3.fromRGB(255,255,255)
        end
    end
end

-- ========== DRAG ==========
local function makeDraggable(topbar, main)
    local dragging, dragInput, dragStart, startPos
    topbar.InputBegan:Connect(function(input)
        if St.guiLocked then return end
        if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
            dragging=true; dragStart=input.Position; startPos=main.Position
            input.Changed:Connect(function()
                if input.UserInputState==Enum.UserInputState.End then dragging=false end
            end)
        end
    end)
    topbar.InputChanged:Connect(function(input)
        if St.guiLocked then return end
        if input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch then
            dragInput=input
        end
    end)
    local c = UserInputService.InputChanged:Connect(function(input)
        if St.guiLocked then return end
        if input==dragInput and dragging then
            local d = input.Position - dragStart
            main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+d.X, startPos.Y.Scale, startPos.Y.Offset+d.Y)
        end
    end)
    table.insert(_G.CPL_Connections, c)
end
makeDraggable(U.topBar, U.mainFrame)

local function tryFind(container)
    if not container then return nil end
    for _,name in ipairs(K.KNOWN_NAMES) do
        local r = container:FindFirstChild(name)
        if r and r:IsA("RemoteEvent") then return r end
    end
    for _,c in ipairs(container:GetChildren()) do
        if c:IsA("RemoteEvent") and c.Name:lower():find("block") then return c end
    end
    return nil
end
local function findRemote()
    local rrs
    pcall(function() rrs = game:GetService("RobloxReplicatedStorage") end)
    local r = tryFind(rrs) if r then return r end
    r = tryFind(ReplicatedStorage) if r then return r end
    for _,c in ipairs(game:GetDescendants()) do
        if c:IsA("RemoteEvent") and c.Name:lower():find("block") then return c end
    end
    return nil
end
St.remote = findRemote()

local cachedFullPayload, cachedIdlePayload = nil, nil
local lastPower = -1
local lastPayloadType = "NORMAL"

local function buildPayload(depth, rep)
    local main = {}
    local nested = {{}}
    local cur = nested[1]
    for _ = 1, depth do
        local n = {}
        cur[1] = n
        cur = n
    end
    for _ = 1, rep do main[#main+1] = nested end
    return main
end

local PAYLOAD_MULT = {
    NORMAL = { d = 1.0,  r = 1.0 },
    HIGH   = { d = 1.6,  r = 1.5 },
    MID    = { d = 1.3,  r = 1.25 },
    LOW    = { d = 0.7,  r = 0.7 },
}

local BURST_COUNT = { NORMAL = 1, LOW = 2, MID = 4, HIGH = 8 }

local function getEffectivePayloadType()
    if not cfg.smartPayload then return cfg.payloadType end
    if K.isMobile and cfg.smartPayload and cfg.smartBurst then
        return "LOW"
    end
    local p = St.currentPing
    if p < 200 then return "HIGH"
    elseif p < 600 then return "MID"
    elseif p < 1200 then return "NORMAL"
    else return "LOW" end
end
_G.CPL_getEffectivePayloadType = getEffectivePayloadType

local function getEffectiveBurstType()
    if not cfg.smartBurst then return cfg.burstMode end
    local p = St.currentPing
    if K.isMobile then
        if p < 200 then return "MID"
        elseif p < 600 then return "LOW"
        else return "NORMAL" end
    end
    if p < 200 then return "HIGH"
    elseif p < 600 then return "MID"
    elseif p < 1200 then return "NORMAL"
    else return "LOW" end
end
_G.CPL_getEffectiveBurstType = getEffectiveBurstType

local function getPayload(isCarryingNow)
    local effectiveType = getEffectivePayloadType()
    if lastPower ~= cfg.power or lastPayloadType ~= effectiveType then
        cachedFullPayload, cachedIdlePayload = nil, nil
        lastPower = cfg.power
        lastPayloadType = effectiveType
    end
    local mult = PAYLOAD_MULT[effectiveType] or PAYLOAD_MULT.NORMAL

    if isCarryingNow then
        if not cachedFullPayload then
            local baseDepth = K.isMobile and 42 or 150
            local baseRep   = math.min(math.floor(cfg.power / (K.isMobile and 70 or 200)), K.isMobile and 2200 or 8000)
            local depth = math.floor(baseDepth * mult.d)
            local rep   = math.floor(baseRep * mult.r)
            cachedFullPayload = buildPayload(depth, rep)
        end
        return cachedFullPayload
    else
        if not cachedIdlePayload then
            local ip    = math.min(cfg.power, 30000)
            local baseDepth = K.isMobile and 30 or 100
            local baseRep   = math.min(math.floor(ip / (K.isMobile and 90 or 260)), K.isMobile and 1200 or 4000)
            local depth = math.floor(baseDepth * mult.d)
            local rep   = math.floor(baseRep * mult.r)
            cachedIdlePayload = buildPayload(depth, rep)
        end
        return cachedIdlePayload
    end
end

local loopId = 0
local function runPingLoop()
    loopId += 1
    local myId = loopId
    local failCount = 0
    while St.active and loopId == myId do
        if not St.remote then
            St.remote = findRemote()
            if not St.remote then task.wait(0.5) continue end
        end
        local carrying = isCarrying()
        local payload = getPayload(carrying)
        local burstCount = BURST_COUNT[getEffectiveBurstType()] or 1
        local ok = true
        for _ = 1, burstCount do
            local success = pcall(function() St.remote:FireServer(payload) end)
            if not success then ok = false; break end
        end
        if ok then failCount = 0
        else
            failCount += 1
            if failCount >= 3 then
                St.remote = findRemote()
                failCount = 0
                task.wait(0.3)
            end
        end
        local delay = cfg.interval
        if St.currentPing > 1400 then delay = delay * 2
        elseif St.currentPing > 1000 then delay = delay * 1.5 end
        task.wait(delay)
    end
end

local function flipLag(state, isManual)
    if isManual then St.manualOverride = state end
    if St.active == state then return end
    St.active = state
    local theme = K.THEMES[cfg.themeIndex] or K.THEMES[8]
    if St.active then
        U.activateBtn.Text, U.activateBtn.TextColor3 = "DEACTIVATE", Color3.fromRGB(0,0,0)
        U.activateBtn.BackgroundColor3 = theme.main
        if not St.remote then St.remote = findRemote() end
        if St.remote then task.spawn(runPingLoop) end
    else
        U.activateBtn.Text, U.activateBtn.TextColor3 = "ACTIVATE", Color3.fromRGB(255,255,255)
        tw(U.activateBtn, {BackgroundColor3 = Color3.fromRGB(15,15,15)}, 0.2)
    end
end

local inputConn = UserInputService.InputBegan:Connect(function(input, processed)
    local kc = input.KeyCode
    if kc == Enum.KeyCode.Unknown then return end
    local isGp = isGamepad(kc)
    local isKb = input.UserInputType == Enum.UserInputType.Keyboard
    if St.listeningFor then
        if kc == Enum.KeyCode.Escape then
            St.listeningFor = nil
            U.kbBtn.Text, U.gpBtn.Text = cfg.keybindKb, cfg.keybindGp
            return

        if St.listeningFor == "kb" and isKb and not K.BLACKLISTED[kc] then
            cfg.keybindKb = kc.Name St.listeningFor = nil U.kbBtn.Text = kc.Name saveConfig() return
        end
        if St.listeningFor == "gp" and isGp then
            cfg.keybindGp = kc.Name St.listeningFor = nil U.gpBtn.Text = kc.Name saveConfig() return
        end
        return
    end
    if processed then return end
    local kbEnum = resolveKb(cfg.keybindKb)
    local gpEnum = resolveKb(cfg.keybindGp)
    if (kbEnum and kc == kbEnum and isKb) or (gpEnum and kc == gpEnum and isGp) then
        if tick() - lastInputTick > 0.2 then
            lastInputTick = tick()
            flipLag(not St.active, true)
        end
    end
end)

local function introSetupEvents(IU, IS, closeIntro)
    TweenService:Create(IU.card, TweenInfo.new(0.6, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
        Size = UDim2.new(0, 460, 0, 225)
    }):Play()
    task.delay(0.12, function()
        TweenService:Create(IU.avatar, TweenInfo.new(0.45, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 52, 0, 52)
        }):Play()
    end)
    task.delay(0.2, function()
        TweenService:Create(IU.mainTitle, TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Position = UDim2.new(0, 82, 0, 16)
        }):Play()
        TweenService:Create(IU.welcomeText, TweenInfo.new(0.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Position = UDim2.new(0, 82, 0, 52),
            TextTransparency = 0
        }):Play()
    end)
    IU.skipBtn.MouseEnter:Connect(function()
        TweenService:Create(IU.skipBtn, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(50, 25, 30),
            Size = UDim2.new(0, 120, 0, 30)
        }):Play()
        IU.skipStroke.Thickness = 1.8
    end)
    IU.skipBtn.MouseLeave:Connect(function()
        TweenService:Create(IU.skipBtn, TweenInfo.new(0.2), {
            BackgroundColor3 = Color3.fromRGB(25, 18, 22),
            Size = UDim2.new(0, 116, 0, 28)
        }):Play()
        IU.skipStroke.Thickness = 1.3
    end)
    IU.skipBtn.MouseButton1Click:Connect(closeIntro)
    task.spawn(function()
        while not IS.isClosing do
            IU.pulseRing.Size = UDim2.new(0, 52, 0, 52)
            IU.pulseStroke.Transparency = 0.2
            local pT = TweenService:Create(IU.pulseRing, TweenInfo.new(1.6, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), { Size = UDim2.new(0, 78, 0, 78) })
            local fT = TweenService:Create(IU.pulseStroke, TweenInfo.new(1.6, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), { Transparency = 1 })
            pT:Play(); fT:Play()
            task.wait(1.8)
        end
    end)
    IS.renderConn = RunService.RenderStepped:Connect(function()
        if IS.isClosing then return end
        local t = tick()
        local loudness = 0
        pcall(function() loudness = IS.sound.PlaybackLoudness end)
        local beat = 0
        if loudness and loudness > 15 then beat = math.clamp((loudness - 30) / 320, 0, 1)
        else beat = math.clamp(math.abs(math.sin(t * 7.2)) ^ 2.8, 0, 1) end
        IU.mainTitle.TextTransparency = (1 - beat) * 0.35
        IU.titleStroke.Thickness = 1.8 + (beat * 1.0)
        IU.strokeGrad.Rotation = (t * 60) % 360
        IU.auraGradient.Rotation = (t * 90) % 360
        IU.imgShade.BackgroundTransparency = 0.72 + math.sin(t * 1.5) * 0.04
        local tip = 12 + math.sin(t * 8) * 3
        IU.barTipGlow.Size = UDim2.new(0, tip, 0, tip)
        local b4 = beat * 4
        IU.musicIcon.Size = UDim2.new(0, 18 + b4, 0, 18 + b4)
        IU.musicIcon.Position = UDim2.new(0, 12 - (b4 / 2), 0.5, -9 - (b4 / 2))
        IU.songStroke.Thickness = 1.2 + (b4 / 4)
        for i, b in ipairs(IU.eqBars) do
            local val = math.clamp(math.abs(math.sin(t * 8 + (i * 1.5))) * 16 * math.max(beat, 0.4), 3, 16)
            b.Size = UDim2.new(0, 3, 0, val)
            b.BackgroundColor3 = Color3.fromRGB(255, math.floor(140 + (val / 16) * 65), math.floor(25 + (val / 16) * 30))
        end
    end)
end

local function introClose(IU, IS, finishOnce)
    if IS.isClosing then return end
    IS.isClosing = true
    if IS.renderConn then IS.renderConn:Disconnect() end
    task.spawn(function()
        local ff = TweenInfo.new(0.35, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        pcall(function() TweenService:Create(IS.sound, ff, { Volume = 0 }):Play() end)
        pcall(function() TweenService:Create(IU.blur, ff, { Size = 0 }):Play() end)
        pcall(function() TweenService:Create(IU.overlay, ff, { BackgroundTransparency = 1 }):Play() end)
        local cY = TweenService:Create(IU.card, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
            Size = UDim2.new(0, 460, 0, 3), GroupTransparency = 0.15
        })
        cY:Play()
        task.wait(0.08)
        finishOnce()
        cY.Completed:Wait()
        local cX = TweenService:Create(IU.card, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.In), {
            Size = UDim2.new(0, 0, 0, 3), GroupTransparency = 1
        })
        cX:Play()
        cX.Completed:Wait()
        pcall(function() IS.sound:Stop(); IS.sound:Destroy() end)
        pcall(function() IU.blur:Destroy() end)
        pcall(function() IU.introGui:Destroy() end)
    end)
end

local function introRunProgress(IU, IS, closeIntro)
    pcall(function()
        IS.sound.TimePosition = Intro.StartSecond
        IS.sound:Play()
    end)
    TweenService:Create(IU.barFill, TweenInfo.new(Intro.Duration, Enum.EasingStyle.Linear), {
        Size = UDim2.new(1, 0, 1, 0)
    }):Play()
    local startTime = tick()
    while true do
        local elapsed = tick() - startTime
        local alpha = math.clamp(elapsed / Intro.Duration, 0, 1)
        IU.percentLabel.Text = "%" .. tostring(math.floor(alpha * 100))
        if alpha < 0.55 then IU.statusLabel.Text = "Loading Your Script..."
        elseif alpha < 0.90 then IU.statusLabel.Text = "Almost there!"
        else
            IU.statusLabel.Text = "Ready! ✨"
            IU.statusLabel.TextColor3 = Color3.fromRGB(255, 185, 50)
            IU.percentLabel.TextColor3 = Color3.fromRGB(255, 185, 50)
        end
        if alpha >= 1 or IS.isClosing then break end
        task.wait(0.03)
    end
    task.wait(0.2)
    closeIntro()
end

local function playIntro(onComplete)
    local finished = false
    local function finishOnce()
        if finished then return end
        finished = true
        if onComplete then task.spawn(onComplete) end
    end
    local IU = {}
    local IS = { isClosing = false, sound = nil, renderConn = nil }
    local ok, err = pcall(function()
        introBuildUI(IU, IS)
        local closeIntro
        closeIntro = function() introClose(IU, IS, finishOnce) end
        introSetupEvents(IU, IS, closeIntro)
        introRunProgress(IU, IS, closeIntro)
    end)
    if not ok then
        warn("[Intro] Error: " .. tostring(err))
        finishOnce()
    end
end

task.spawn(function()
    local ok, err = pcall(function() playIntro(showMainUI) end)
    if not ok then
        warn("[Intro] Fatal: " .. tostring(err))
        showMainUI()
    end
end)
