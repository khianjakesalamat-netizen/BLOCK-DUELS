local function parentGui(gui)
	local function tryParent(target)
		pcall(function() gui.Parent = target end)
		return gui.Parent ~= nil
	end
	if tryParent(pg) then return true end
	pcall(function()
		if typeof(gethui) == "function" then
			local h = gethui()
			if h and tryParent(h) then return true end
		end
	end)
	if tryParent(game:GetService("CoreGui")) then return true end
	return false
end

local function startAntiBat()
	local char = LP.Character
	if not char then return end
	local root = char:FindFirstChild("HumanoidRootPart")
	if not root then return end
	if AntiBatConn then AntiBatConn:Disconnect() end
	AntiBatConn = RunService.Heartbeat:Connect(function()
		if not AntiBatEnabled then return end
		if not root or not root.Parent then return end
		local origXZ = Vector3.new(root.Velocity.X, 0, root.Velocity.Z)
		root.Velocity = Vector3.new(4000, root.Velocity.Y, 4000)
		RunService.RenderStepped:Wait()
		if root and root.Parent then
			root.Velocity = Vector3.new(origXZ.X, root.Velocity.Y, origXZ.Z)
		end
	end)
end

local function stopAntiBat()
	if AntiBatConn then
		AntiBatConn:Disconnect()
		AntiBatConn = nil
	end
end

UIS.JumpRequest:Connect(function()
	if not InfiniteJumpEnabled then return end
	local char = LP.Character
	if not char then return end
	local root = char:FindFirstChild("HumanoidRootPart")
	if root then
		root.Velocity = Vector3.new(root.Velocity.X, 57, root.Velocity.Z)
	end
end)

local function startAntiRagdoll()
	if AntiRagdollConn then return end
	AntiRagdollConn = RunService.Heartbeat:Connect(function()
		local char = LP.Character
		if not char then return end
		local hum2 = char:FindFirstChildOfClass("Humanoid")
		local root = char:FindFirstChild("HumanoidRootPart")
		if hum2 then
			local st = hum2:GetState()
			if st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown then
				hum2:ChangeState(Enum.HumanoidStateType.Running)
				workspace.CurrentCamera.CameraSubject = hum2
				pcall(function()
					local pm = LP.PlayerScripts:FindFirstChild("PlayerModule")
					if pm then require(pm:FindFirstChild("ControlModule")):Enable() end
				end)
				if root then
					root.Velocity = Vector3.new(0, 0, 0)
					root.RotVelocity = Vector3.new(0, 0, 0)
				end
			end
		end
		for _, obj in ipairs(char:GetDescendants()) do
			if obj:IsA("Motor6D") and not obj.Enabled then
				obj.Enabled = true
			end
		end
	end)
end

local function setToggleVisual(pill, knob, on)
	TS:Create(knob, TweenInfo.new(0.15), {
		Position = on and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9),
		BackgroundColor3 = on and Color3.fromRGB(0, 255, 140) or Color3.fromRGB(255, 255, 255),
	}):Play()
	TS:Create(pill, TweenInfo.new(0.15), {
		BackgroundColor3 = on and Color3.fromRGB(0, 90, 60) or Color3.fromRGB(50, 50, 58),
	}):Play()
end

local SCALE_MIN, SCALE_MAX = 0.55, 1.35
local function applyScale(alpha)
	alpha = math.clamp(alpha, 0, 1)
	guiScale = SCALE_MIN + (SCALE_MAX - SCALE_MIN) * alpha
	uiScaleObj.Scale = guiScale
	ScaleValue.Text = string.format("%.2fx", guiScale)
	SliderFill.Size = UDim2.new(alpha, 0, 1, 0)
	SliderKnob.Position = UDim2.new(alpha, 0, 0.5, 0)
end
applyScale((guiScale - SCALE_MIN) / (SCALE_MAX - SCALE_MIN))

local sliding = false
local function updateSliderFromInput(pos)
	local abs = SliderTrack.AbsolutePosition
	local size = SliderTrack.AbsoluteSize
	if size.X <= 0 then return end
	local a = (pos.X - abs.X) / size.X
	applyScale(a)
end

SliderKnob.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		sliding = true
	end
end)
SliderTrack.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		sliding = true
		updateSliderFromInput(input.Position)
	end
end)
UIS.InputChanged:Connect(function(input)
	if not sliding then return end
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		updateSliderFromInput(input.Position)
	end
end)
UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		sliding = false
	end
end)

do
	local dragging, start, startPos
	Frame.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			start = input.Position
			startPos = Frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	UIS.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			local d = input.Position - start
			Frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
end
