-- ============================================
-- CONFIG (save / load)
-- ============================================
local CONFIG_FILE = "AspectAntiBat.json"

local config = {
    antiBat = false,
    infJump = false,
    posX    = nil,
    posY    = nil,
}

local canSave = (typeof(writefile) == "function")
    and (typeof(readfile) == "function")
    and (typeof(isfile) == "function")

local function saveConfig()
    if not canSave then return end
    pcall(function()
        writefile(CONFIG_FILE, HttpService:JSONEncode(config))
    end)
end

local function loadConfig()
    if not canSave then return end
    pcall(function()
        if not isfile(CONFIG_FILE) then return end
        local data = HttpService:JSONDecode(readfile(CONFIG_FILE))
        if type(data) ~= "table" then return end
        config.antiBat = data.antiBat == true
        config.infJump = data.infJump == true
        config.posX    = tonumber(data.posX)
        config.posY    = tonumber(data.posY)
    end)
end

loadConfig()

-- ============================================
-- LOGICA: ANTI BAT
-- ============================================
local AntiBatConn = nil
local Speed = 10000
local Angle = 0
local Direction = 1

local function startAntiBat()
    local char = LocalPlayer.Character
    if not char then return end

    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    if AntiBatConn then
        AntiBatConn:Disconnect()
    end

    AntiBatConn = RunService.Heartbeat:Connect(function()
        if not root or not root.Parent then return end

        local origXZ = Vector3.new(root.Velocity.X, 0, root.Velocity.Z)

        Speed = Speed + (50 * Direction)
        if Speed >= 20000 then
            Speed = 20000
            Direction = -1
        elseif Speed <= 10000 then
            Speed = 10000
            Direction = 1
        end

        Angle = Angle + (math.random() * 0.5)

        local radius = Speed + math.random(-5000, 5000)
        local newX = math.cos(Angle) * radius
        local newZ = math.sin(Angle) * radius

        root.Velocity = Vector3.new(newX, root.Velocity.Y, newZ)

        RunService.RenderStepped:Wait()

        root.Velocity = Vector3.new(origXZ.X, root.Velocity.Y, origXZ.Z)
    end)
end

local function stopAntiBat()
    if AntiBatConn then
        AntiBatConn:Disconnect()
        AntiBatConn = nil
    end
    Speed = 10000
    Angle = 0
    Direction = 1
end

local function setAntiBat(on)
    config.antiBat = on and true or false
    if config.antiBat then startAntiBat() else stopAntiBat() end
    saveConfig()
end

-- ============================================
-- LOGICA: INFINITE JUMP
-- ============================================
local JUMP_POWER = 55
local jumpHeld   = false

local function applyJump()
    if not config.infJump then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    root.Velocity = Vector3.new(root.Velocity.X, JUMP_POWER, root.Velocity.Z)
end

UIS.JumpRequest:Connect(applyJump)

UIS.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.Space then jumpHeld = true end
end)

UIS.InputEnded:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.Space then jumpHeld = false end
end)

RunService.Heartbeat:Connect(function()
    if config.infJump and jumpHeld then applyJump() end
end)

LocalPlayer.CharacterAdded:Connect(function(char)
    jumpHeld = false
    if config.antiBat then
        char:WaitForChild("HumanoidRootPart", 10)
        task.wait(0.2)
        startAntiBat()
    end
end)

-- DRAG SYSTEM + SALVATAGGIO POSIZIONE
-- ============================================
do
    local dragging = false
    local dragStart, startPos, dragInput

    local function updateDrag(input)
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end

    Main.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    -- salva posizione
                    config.posX = Main.Position.X.Offset
                    config.posY = Main.Position.Y.Offset
                    Main.Position = UDim2.new(0, config.posX, 0, config.posY)
                    saveConfig()
                end
            end)
        end
    end)

    Main.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            updateDrag(input)
        end
    end)
end

-- ============================================
-- TOGGLE ANIMATION HELPER
-- ============================================
local TWEEN_INFO = TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
local COLOR_ON = Color3.fromRGB(147, 51, 234)
local COLOR_OFF = Color3.fromRGB(55, 55, 68)
local KNOB_ON = UDim2.new(1, -21, 0.5, -9)
local KNOB_OFF = UDim2.new(0, 3, 0.5, -9)

local function setToggle(track, knob, state)
    TweenService:Create(track, TWEEN_INFO, {
        BackgroundColor3 = state and COLOR_ON or COLOR_OFF
    }):Play()
    TweenService:Create(knob, TWEEN_INFO, {
        Position = state and KNOB_ON or KNOB_OFF
    }):Play()
end
